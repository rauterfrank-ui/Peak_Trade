                mod = "." * node.level + mod
            mod_imports.append(mod)
        elif isinstance(node, ast.Assign):
            for t in node.targets:
                if isinstance(t, ast.Name) and t.id.isupper():
                    try:
                        val = ast.literal_eval(node.value)
                    except Exception:
                        val = ast.get_source_segment(text, node.value) or "NON_LITERAL"
                    params_by_file[mf.name].append((t.id, val))
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            objects.append({
                "OBJECT_ID": f"FUNC:{mf.stem}.{node.name}",
                "OBJECT_TYPE": "FUNCTION",
                "SOURCE_PATH": rel(mf),

                "PURPOSE": (ast.get_docstring(node) or "")[:500],
                "CURRENT_STATUS": "PROVEN_CURRENT",
