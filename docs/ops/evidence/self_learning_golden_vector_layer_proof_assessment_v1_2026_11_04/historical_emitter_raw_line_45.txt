def rel(p):
    try:
        return str(p.relative_to(REPO))

external_handoffs = []
params_by_file = defaultdict(list)

for mf in modules:
    text = mf.read_text(encoding="utf-8", errors="replace")
    try:
        tree = ast.parse(text)
    except SyntaxError:
        continue
    mod_imports = []
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            for a in node.names:
                mod_imports.append(a.name)
        elif isinstance(node, ast.ImportFrom):
            mod = node.module or ""
            if node.level:
                mod = "." * node.level + mod
            mod_imports.append(mod)
        elif isinstance(node, ast.Assign):
            for t in node.targets:
                if isinstance(t, ast.Name) and t.id.isupper():
                    try:
                        val = ast.literal_eval(node.value)
                    except Exception:
                        val = ast.get_source_segment(text, node.value) or "NON_LITERAL"
                    params_by_file[mf.name].append((t.id, val))
        elif isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            objects.append({
                "OBJECT_ID": f"FUNC:{mf.stem}.{node.name}",
                "OBJECT_TYPE": "FUNCTION",
                "SOURCE_PATH": rel(mf),
                "SYMBOL": node.name,
            })
        elif isinstance(node, ast.ClassDef):
            objects.append({
                "OBJECT_ID": f"CLASS:{mf.stem}.{node.name}",
                "OBJECT_TYPE": "CLASS",
                "SOURCE_PATH": rel(mf),
                "SYMBOL": node.name,
            })
    for imp in mod_imports:
        if imp.startswith("src.") and not imp.startswith(PREFIX):
            external_handoffs.append({
                "SOURCE": rel(mf),
                "TARGET_UNIVERSE": imp.split(".")[1] if len(imp.split(".")) > 1 else imp,
                "CONTRACT": imp,
            })
