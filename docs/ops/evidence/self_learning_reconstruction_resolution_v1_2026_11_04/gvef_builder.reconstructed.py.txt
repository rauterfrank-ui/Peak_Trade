def schema_md() -> str:
    parts = ["## Appendix A — Schema field registry\n"]
    for sname, flist in SCHEMAS.items():
        parts.append(f"### {sname}\n\n| NAME | TYPE | REQ | NULL | SEMANTICS |\n|------|------|-----|------|----------|\n")
        for f in flist:
            parts.append(f"| {f['NAME']} | {f['TYPE']} | {f['REQUIRED']} | {f['NULLABLE']} | {f['SEMANTICS']} |\n")
        parts.append("\n")
    return "".join(parts)


def components_md() -> str:
    lines = ["## Appendix B — Component registry\n"]
    for c in COMPONENTS:
        lines.append(f"### `{c['COMPONENT_ID']}`\n")
        for k in (
            "PURPOSE",
            "AUTHORITY",
            "OWNER",
            "STATEFUL",
            "SIDE_EFFECT_CLASS",
            "FAILURE_BEHAVIOR",
            "CURRENT_PROVENANCE",
        ):
            if k in c:
                lines.append(f"- **{k}:** {c[k]}\n")
        lines.append(f"- **ALLOWED_CALLERS:** {', '.join(c.get('ALLOWED_CALLERS', []))}\n")
        lines.append(f"- **FORBIDDEN_CALLEES:** {', '.join(c.get('FORBIDDEN_CALLEES', []))}\n\n")
    return "".join(lines)
