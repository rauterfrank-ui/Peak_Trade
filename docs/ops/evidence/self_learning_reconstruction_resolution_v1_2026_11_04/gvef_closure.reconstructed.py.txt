def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def load_json(name: str) -> dict:
    return json.loads((PKG / name).read_text(encoding="utf-8"))


def save_json(name: str, obj: object) -> None:
    (PKG / name).write_text(json.dumps(obj, indent=2, sort_keys=True) + "\n", encoding="utf-8")


def fix_fanout_matrix(data: dict) -> None:
    data["enum"] = CANONICAL_FANOUT
    data["fan_out_reproof_resolver_v1"] = {
        "escalation_order": CANONICAL_FANOUT,
        "rule": "max rank in escalation_order across all matched triggers; never downgrade",
        "failure_taxonomy_field": "RESOLVER_TRIGGER_TAGS",
        "blocked_execution_failure_classes": [
            "CURRENT_UNKNOWN",
            "CURRENT_CONFLICT",
            "CURRENT_VIOLATION",
        ],
        "blocked_is_not_fan_out_enum": True,
    }


def fix_failure_taxonomy(data: dict) -> None:
    new_classes = []
    for c in data["classes"]:
        cls = c["class"]
        nc = {k: v for k, v in c.items() if k != "REPROOF_CLASS"}
        nc["FAN_OUT_RESOLVER"] = "gvef.fan_out_reproof_resolver_v1"
        nc["RESOLVER_TRIGGER_TAGS"] = FAILURE_RESOLVER_BINDINGS.get(cls, [])
        if cls.startswith("CURRENT_"):
            nc["EXECUTION_BLOCKED"] = True
            nc["FAN_OUT_ENUM_APPLIES"] = False
        else:
            nc["EXECUTION_BLOCKED"] = False
            nc["FAN_OUT_ENUM_APPLIES"] = True
        new_classes.append(nc)
    data["classes"] = new_classes
    data["resolver_note"] = (
        "Failure class does not embed reproof enum; resolver applies fan_out_reproof_matrix rules."
    )


def fix_contract_schema(data: dict) -> None:
    enum_str = "|".join(CANONICAL_FANOUT)
    blob = json.dumps(data)
    blob = blob.replace("LOCAL|DOWNSTREAM|WHOLE_SYSTEM_REPROOF", enum_str)
    blob = re.sub(r'"LOCAL\|DOWNSTREAM\|WHOLE_SYSTEM_REPROOF"', f'"{enum_str}"', blob)
    fixed = json.loads(blob)
    data.clear()
    data.update(fixed)
    if "digest_model" in data:
        dm = data["digest_model"].get("verdicts", {})
        if isinstance(dm.get("DIGEST_CHANGED_UNEXPECTED"), str):
            dm["DIGEST_CHANGED_UNEXPECTED"] = "fail-closed; WHOLE_SYSTEM_REPROOF via resolver"

def fix_components(data: dict) -> None:
    for c in data["components"]:
        if "EVIDENCE_PRODUCED" in c and "EVIDENCE_OUTPUT" not in c:
            c["EVIDENCE_OUTPUT"] = c.pop("EVIDENCE_PRODUCED")
        if "CURRENT_PROVENANCE" in c and "PROVENANCE" not in c:
            c["PROVENANCE"] = c["CURRENT_PROVENANCE"]
        for k in (
            "FORBIDDEN_CALLERS",
            "FORBIDDEN_CALLEES",
            "ALLOWED_CALLERS",
            "ALLOWED_CALLEES",
        ):
            c.setdefault(k, [])
        c.setdefault("READ_BOUNDARIES", [])
        c.setdefault("WRITE_BOUNDARIES", [])

def scan_alias_count(root: Path) -> int:
    aliases = 0
    patterns = [
        re.compile(r'"(LOCAL|DOWNSTREAM)"(?!_)'),
        re.compile(r"LOCAL\|DOWNSTREAM\|"),
        re.compile(r"DOWNSTREAM or WHOLE"),
        re.compile(r"\bREPROOF_CLASS\b"),
    ]
    for p in root.iterdir():
        if p.suffix in {".json", ".md", ".txt"}:
            t = p.read_text(encoding="utf-8", errors="ignore")
            for pat in patterns:
                aliases += len(pat.findall(t))
    return aliases


def parity_check() -> dict:
    conflicts = []
    arch = load_json("architecture_manifest.json")
    comp = load_json("component_manifest.json")
    if len(comp["components"]) != 23:
        conflicts.append(f"component_count={len(comp['components'])}")
    evaluators = arch.get("evaluators") or [
        "ranking_universe_evaluator_v1",
        "productive_trading_path_evaluator_v1",
        "self_learning_evaluator_v1",
        "optimization_universe_evaluator_v1",
        "market_intelligence_evaluator_v1",
    ]
    if len(evaluators) != 5:
        conflicts.append("evaluator_count!=5")
    alias = scan_alias_count(PKG)
    if alias > 0:
        conflicts.append(f"fan_out_alias_count={alias}")
    ft = load_json("failure_taxonomy.json")
    for c in ft["classes"]:
        if "REPROOF_CLASS" in c:
            conflicts.append("failure_has_REPROOF_CLASS")
        if c.get("REPROOF_CLASS"):
            conflicts.append("failure_has_REPROOF_CLASS_field")
    steps = [t.get("to") for t in arch["pipeline_transitions"]]
    if "DELTA_MANIFEST_BUILD" not in steps:
        conflicts.append("missing_DELTA_MANIFEST_BUILD")
    md = (PKG / "Peak_Trade_Golden_Vector_Evaluation_Fabric_Blueprint_V1_5.md").read_text(encoding="utf-8")
    for label in "ABCDEFGH":
        if f"### {label}." not in md:
            conflicts.append(f"missing_diagram_{label}")
    if "See section 3 pipeline" in md:
        conflicts.append("diagram_C_placeholder")
    return {"conflicts": conflicts, "conflict_count": len(conflicts)}
