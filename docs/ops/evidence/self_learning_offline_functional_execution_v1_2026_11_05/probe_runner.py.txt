    (OBS/(ident+'.json')).write_text(json.dumps(dict(criterion_id=ident,criterion_status=status,observed=observed,limitation=limitation),indent=2,ensure_ascii=False,default=str)+'\n')

def baseline_case(fragment):
    cases=[c for c in ET.parse(OUT/'baseline.xml').getroot().iter('testcase') if fragment in c.attrib['name']]
    assert cases and all(c.find('failure') is None and c.find('error') is None and c.find('skipped') is None for c in cases)

class SecondCandidateChange(DeterministicDomainEvaluatorV1):
    mode='verdict'
    second=None
    def evaluate_candidate(self,*,context,replay):
        result=super().evaluate_candidate(context=context,replay=replay)
        if self.candidate_calls==2:
            if self.mode=='verdict':
                result=result.model_copy(update={'verdict':DomainVerdict.FAIL,'failure_classification':FailureClassification.INVARIANT_FAILURE})
            elif self.mode=='metric':
                result=result.model_copy(update={'metrics':[MetricResultV1(metric_id='m.candidate',value=999)]})
            else:
                result=result.model_copy(update={'semantic_digest_deltas':{'probe':'second-candidate-changed'}})
            self.second=result
        return result

class BaselineDigestDrift(DeterministicDomainEvaluatorV1):
    def evaluate_baseline(self,*,context,replay):
        result=super().evaluate_baseline(context=context,replay=replay)
        return result.model_copy(update={'metrics':[MetricResultV1(metric_id='m.baseline',value=777)]}) if self.baseline_calls==2 else result

class ChangingReplay(DeterministicReplayAdapterV1):
    calls=0
    def replay(self,*,run,context):
        self.calls+=1
        if self.calls==1:return context.replay_trace
        return ReplayTraceV1.model_validate(replay_trace([*context.replay_trace.entries,{'offline_probe_generation':self.calls}]))
